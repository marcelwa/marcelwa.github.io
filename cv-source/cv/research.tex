\cvsection{Research Interests}


%-------------------------------------------------------------------------------
%	CONTENT
%-------------------------------------------------------------------------------
\begin{cvparagraph}
\begin{itemize}[leftmargin=2ex, nosep, noitemsep]
  \renewcommand{\labelitemi}{\textcolor{awesome}{\textbullet}}
  \item \textbf{Design Automation for Atomic-Scale Computing:} full-stack design flows for emerging post-CMOS substrates like Silicon Dangling Bonds; compiling conventional RTL specifications down to verified, defect-aware physical layouts of individual atoms.
  \item \textbf{Logic Synthesis \& Machine Learning:} bridging modern AI and classical logic synthesis; developing open-source frameworks to train machine-learning models for logic optimization and technology mapping.
  \item \textbf{Physical Design for Photonic \& Quantum Circuits:} adapting planar, crossing-aware layout algorithms to emerging computing substrates including silicon photonics and superconducting qubits.
  \item \textbf{HPC \& Quantum Systems Integration:} integrating quantum accelerators into high-performance computing infrastructure (HPCQC) and scaling design automation algorithms for distributed execution.
\end{itemize}
\end{cvparagraph}
