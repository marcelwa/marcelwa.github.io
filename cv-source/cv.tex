\newcommand{\makeauthorbold}[1]{%

    \ifthenelse{\equal{\namepartfamily}{#1}}%
    {\textbf{\ifblank{\namepartgiven}{}{\namepartgiven\space}\namepartfamily}}%
    {\ifblank{\namepartgiven}{}{\namepartgiven\space}\namepartfamily}%
    \ifthenelse{\value{listcount}<\value{liststop}}%
    {\addcomma\space}%
    {}%
    \ifthenelse{\value{listcount}=\value{liststop}\AND\ifmorenames}%
    {\addcomma\space\bibstring{andothers}}%
    {}%          
  }%
}
