% the two rows are written out directly here instead of via that macro.
% Styled like a normal \cventry's position/date + description rows (rather
% than the smaller \cvsubentry style) so the students stay visually on par
% with every other entry in the CV; only the repeated institution row is
% dropped.
\newcommand{\phdsubentry}[3]{%
  \vspace{-0.5mm}
  \setlength\tabcolsep{0pt}
  \setlength{\extrarowheight}{0pt}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} L{\textwidth - 4.5cm} R{4.5cm}}
    \entrypositionstyle{#1} & \entrydatestyle{#2} \\
    \multicolumn{2}{L{\textwidth}}{\descriptionstyle{#3}}
  \end{tabular*}%
}
